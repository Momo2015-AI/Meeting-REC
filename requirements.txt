# Python 依赖
numpy>=2.0
cffi>=2.0
soundcard>=0.4.6

# SenseVoice 引擎（可选，[asr] engine="sensevoice" 时需要）
#
# ★ 版本与 Python ABI 的硬约束（2026-10-27 实测）：
#   onnxruntime==1.20.1 只发布到 cp313（Python 3.13），**无 cp314 轮子**；
#   cp314 的 onnxruntime（1.24.1+）加载 pybind11 扩展会失败：
#     ImportError: DLL load failed ... 动态链接库(DLL)初始化例程失败（WinError 1114）
#   → 用 SenseVoice 引擎必须用 Python ≤ 3.13（推荐 3.13 + onnxruntime 1.20.1）。
#   → Python 3.14 请保持 [asr] engine = "vosk"（只依赖 models/lib/libvosk.dll，无 ABI 约束）。
#
# ★ funasr-onnx 声明 numpy<=1.26.4，但 numpy 1.26.4 无 cp313 轮子；
#   实际安装用 numpy==2.2.6，并对 funasr-onnx 用 --no-deps 绕过该约束（运行时无兼容问题）。
#
#   .venv\Scripts\python -m pip install numpy==2.2.6 onnxruntime==1.20.1 \
#       sentencepiece==0.2.1 cffi soundcard PyYAML jieba kaldi-native-fbank librosa scipy
#   .venv\Scripts\python -m pip install --no-deps funasr-onnx==0.4.3
funasr-onnx==0.4.3
onnxruntime==1.20.1        # 钉 1.20.1：≥1.21 的 pybind11 扩展在本机加载崩溃(0xC0000005)；1.18量化器有onnx兼容bug；无 cp314 轮子
sentencepiece==0.2.1       # 0.2.2 加载崩溃；0.2.0 无 Windows 轮子
modelscope                 # 模型下载用（scripts/download_sensevoice.py）

# 说明：
# 1. 识别引擎不使用 vosk Python 包，而是通过 ctypes 直绑 models/lib/libvosk.dll，
#    以规避 vosk 包在较新 Python 版本上依赖链不完整导致的导入失败。
# 2. soundcard 的 Windows 轮子不含 cffi 依赖，需单独安装 cffi。
# 3. vendor/ 目录内置了 soundcard 与 pycparser 的离线轮子，
#    可用 pip install --no-deps vendor/*.whl 离线安装。
